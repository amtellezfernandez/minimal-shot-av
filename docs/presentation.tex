\begin{frame}{The Problem — Continuous Policy Collapse in Long-Tail}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Imitation learning} minimises:
      $\;\min_{f}\;\mathbb{E}_{(x,y)\sim\mathcal{D}}\bigl[\mathcal{L}(f(x),y)\bigr]$

      \vspace{0.3em}
      Works on the common driving distribution $\mathcal{D}$.
      \textbf{Fails when $P_\text{test}\neq P_\text{train}$}: continuous distributions
      spread — the policy assigns probability mass to unsafe intermediate states
      that never appeared in training.

      \vspace{0.3em}
      \textbf{WOD-E2E} targets \textbf{11 scenario types} occurring
      $<$0.03\% of driving time: wrong-way actors, animals, debris,
      construction, cut-ins, cyclists.

      \vspace{0.3em}
      \begin{alertblock}{}
        \small Continuous baseline: no world-state reasoning.
        Drives directly into the actor. Gauntlet pass rate: \bad{2.1\%}, collision: \bad{20.5\%}.
      \end{alertblock}
    \end{column}
    \begin{column}{0.47\textwidth}
      \includegraphics[width=\linewidth]{images/kf_baseline_spotlight.png}
      \vspace{0.1em}
      \small\textit{Wrong-way actor, night. Continuous baseline: no awareness the path is
      occupied. Policy variance under OOD → collision.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Three Claims}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\small \textbf{Claim 1 — Discrete Bottleneck Hypothesis}}
        \small Quantizing the action space to \textbf{9 ManeuverTokens} is a

      \end{block}

      \vspace{0.2em}

        \small Cosmos 64d embeddings carry preference signal inaccessible to
        linear models but extractable by a GPU MLP.
        Generation-oriented VLAs fail on discrimination tasks.\\
        \footnotesize\textit{Evidence: linear head $\to$ no gain; GPU MLP $\to$ +0.042 RFS.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\small \textbf{Claim 3 — Causal Bottleneck Analysis}}
        \small ManeuverToken selection is causally determined by geometric
        pressure scalars — no demonstration data, no spurious correlations.
        This precisely localizes the 1.419 RFS oracle gap to the
        visual-to-geometric bridge.\\
        \footnotesize\textit{Evidence: failure modes trace to scalar interactions;
        oracle gap not in action space or ranker.}
      \end{block}
    \end{column}
  \end{columns}

% ── SLIDE 4: 6 World-State Scalars ───────────────────────────────────────────
\begin{frame}{Architecture: Six World-State Scalars}
  All signals computed from raw obstacle positions and route geometry.
  \textbf{Nothing is learned; nothing is scene-specific.}

  \vspace{0.4em}
  \begin{center}
  \small
  \begin{tabular}{llc}
    \toprule
    \textbf{Signal} & \textbf{Definition} & \textbf{Range} \\
    \midrule
    \code{obstacle\_pressure}      & $p = \mathrm{clip}\!\left(\tfrac{10-d_{\min}}{10},0,1\right)$
                                   & $[0,1]$ \\[4pt]
    \code{route\_blockage}         & fraction of forward corridor occupied & $[0,1]$ \\[4pt]
    \code{corridor\_blocked}       & hard blockage within stopping distance & bool \\[4pt]
    \code{left\_clearance}         & free lateral space to the left (m) & ${\geq}0$ \\[4pt]
    \code{right\_clearance}        & free lateral space to the right (m) & ${\geq}0$ \\[4pt]
    \code{preferred\_escape\_side} & $\arg\max(\mathrm{left},\mathrm{right})$ & L / R \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.2em}
  \small $d_{\min}$ = distance to nearest obstacle. Pressure = 1.0 at contact, 0 beyond 10\,m.
  No object identity, no semantic labels, no history. All O(1) to compute.
  Formal invariance tested: $M(s)=M(T(s))$ for any object-identity relabelling $T$.
\end{frame}

% ── SLIDE 5: 9 ManeuverTokens (Discrete Bottleneck) ──────────────────────────
\begin{frame}{Architecture: Nine ManeuverTokens — The Discrete Bottleneck}
  Each token: \textbf{20-point, 5\,s trajectory at 4\,Hz} via Hermite smoothstep.
  Evasive maneuvers apply the lateral offset in the first third for fast separation.

  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \footnotesize
      \begin{tabular}{p{2.5cm}p{3.0cm}cc}
        \toprule
        \textbf{Maneuver} & \textbf{Intent} & \textbf{Speed} & \textbf{Lateral} \\
        \midrule
        \code{stop}           & Emergency stop      & 0.00$\times$ & 0.0\,m \\
        \code{crawl}          & Very slow progress  & 0.15$\times$ & 0.0\,m \\
        \code{maintain}       & Hold current speed  & 1.00$\times$ & 0.0\,m \\
        \code{slow\_yield}    & Decelerate \& yield & 0.50$\times$ & 0.0\,m \\
        \code{nudge\_left}    & Minor left shift    & 0.80$\times$ & $+$0.9\,m \\
        \code{nudge\_right}   & Minor right shift   & 0.80$\times$ & $-$0.9\,m \\
        \code{evasive\_left}  & Strong left evasion & 0.70$\times$ & $+$2.2\,m \\
        \code{evasive\_right} & Strong right evasion& 0.70$\times$ & $-$2.2\,m \\
        \code{lane\_recover}  & Return to centre    & 0.90$\times$ & 0.0\,m \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.35\textwidth}
      \begin{block}{\small Why discrete?}
        \small A continuous policy assigns mass to unsafe \emph{intermediate states}
        between tokens. The token constraint eliminates those states by construction —
        uncertainty changes \textbf{scores}, not physical footprints.
      \end{block}
      \vspace{0.2em}
      \small Coverage: 5 longitudinal $\times$ 4 lateral directions covers all
      safety-critical responses in the Waymo long-tail taxonomy. Nothing missing;
      nothing extra.
    \end{column}
  \end{columns}
\end{frame}

% ── SLIDE 6: Trust-Region Scoring ────────────────────────────────────────────
\begin{frame}{Architecture: Trust-Region Scoring \& Decision Flow}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \textbf{Speed-scaled tolerance:}
      \vspace{-0.2em}
      \[
        s(v)=\mathrm{clip}\!\left(\tfrac{v-1.4}{9.6},0,1\right)\!\times 0.5+0.5
      \]
      \vspace{-0.5em}
      \footnotesize
      \begin{tabular}{lcc}
        \toprule
        Horizon & Lateral & Longitudinal \\
        \midrule
        3\,s & $1.0{\times}s(v)$\,m & $4.0{\times}s(v)$\,m \\
        5\,s & $1.8{\times}s(v)$\,m & $7.2{\times}s(v)$\,m \\
        \bottomrule
      \end{tabular}
      \vspace{0.1em}
      \[
        S(\tau)=0.5\cdot\max_r\,\text{score}_{3s}+0.5\cdot\max_r\,\text{score}_{5s}
      \]
      \small Scores $\leq50$ = unsafe floor. Clearance $<0.55$\,m: $-$1000 penalty.
    \end{column}
    \begin{column}{0.53\textwidth}
      \small\textbf{Decision pipeline (4 steps):}
      \begin{enumerate}\small
        \item Scene $\to$ compute 6 world-state scalars
        \item Reference rule engine maps scalars to expected tokens + scores (76–94)
        \item Each of the 9 tokens scored against references at 3\,s and 5\,s
        \item Token with highest combined score is selected
      \end{enumerate}
      \vspace{0.3em}
      \small Every step emits a full log: maneuver, scores, decision reason,
      all scalar values. In AlpaSim: serialised as \code{reasoning\_text} JSON.
      \textbf{Identical policy binary} in 2D simulator and AlpaSim.
    \end{column}
  \end{columns}
\end{frame}

      \begin{itemize}\small

        \item \textbf{Not the ranker} — gate 7.803 + GPU MLP 7.845 exhaust

          direct policy both fail.

      \small GO\_RIGHT (2.063 regret, $n$=29): temporal gate over-selects.
      Rightward intent requires lateral clearance checks absent from ego-history —
      a geometry-aware camera model would catch this.
