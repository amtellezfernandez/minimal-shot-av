\section{Problem: AVs Fail in Long-Tail Scenarios}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Imitation Learning — What It Optimises}
  Most AV systems are trained by \textbf{imitation learning}: fit a model $f$ that
  mimics the human trajectory at every step.

  \[
    \min_{f} \; \mathbb{E}_{(x,y)\sim\mathcal{D}}\bigl[\mathcal{L}(f(x),\,y)\bigr]
  \]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Works well on:}
      \begin{itemize}
        \item The common driving distribution $\mathcal{D}$
        \item Highway, urban, well-lit, unambiguous situations
        \item Scenarios appearing frequently in training data
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Fails when:}
      \begin{itemize}
        \item $P_\text{test} \neq P_\text{train}$ \quad (covariate shift)
        \item Rare geometry appears at test time
        \item No fallback reasoning when the pattern fails
        \item Trajectory imitation becomes guessing
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.6em}
  \begin{alertblock}{The WOD-E2E long-tail problem}
    Waymo's benchmark targets \textbf{11 scenario types} occurring less than \textbf{0.03\%}
    of driving time: wrong-way actors, animals on road, foreign object debris,
    erratic pedestrians, construction, cut-ins, cyclists, special vehicles.
    A long-tail policy needs a representation that remains meaningful when frequencies change.
  \end{alertblock}
\end{frame}

\begin{frame}{Why Closed-Loop Evaluation Matters}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Open-loop (frame-based):}
      \[
        \text{frame} \xrightarrow{f} \hat{\tau}
      \]
      Tests whether a model produces a plausible trajectory from a snapshot.
      The world is \emph{fixed} — the policy's action has no consequences.

      \vspace{0.4em}
      \textit{A bad action looks fine on paper.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Closed-loop (our evaluation):}
      \begin{align*}
        S_t &\to \text{observation}\; O_t \\
        O_t &\to \pi(O_t) = \tau_t \quad \text{(policy)} \\
        \tau_t &\to \text{control} + \text{physics} \\
            &\quad\; S_{t+1} = f(S_t, \tau_t) \\
        S_{t+1} &\to \text{metrics}
      \end{align*}
    \end{column}
  \end{columns}

  \vspace{0.5em}
  \begin{block}{The key question}
    Can the \textbf{same policy} stay safe as the world changes \emph{because of its
    own decisions}? A wrong evasion at step $t$ creates a worse state at step $t+1$.
    Open-loop benchmarks cannot reveal this failure mode.
  \end{block}
\end{frame}

\begin{frame}{The Baseline Failure}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\linewidth]{images/kf_baseline_spotlight.png}
    \end{column}
    \begin{column}{0.45\textwidth}
      \vspace{0.5em}
      \textbf{Scenario:} wrong-way actor, night, low visibility \\[0.4em]
      \textbf{Policy:} trajectory extrapolation only — no world-state reasoning \\[0.4em]
      \textbf{Result:} drives directly into the actor.
      \begin{itemize}
        \item Does not slow down
        \item Does not deviate
        \item No awareness that the path is occupied
      \end{itemize}
      \vspace{0.6em}
      \begin{alertblock}{}
        This is what happens when a policy has never seen this geometry and has no
        reasoning mechanism to fall back on.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Core Idea: Reason From Geometry, Not Memory}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{The Key Insight}
  \begin{center}
    \Large\textit{``A trajectory that avoids occupied space and respects\\
    route geometry is correct \textbf{regardless of what the obstacle is called}.''} \\[1em]
    \normalsize You don't need to have seen a sheep before to know you should go around it.
  \end{center}

  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{What the policy needs to know:}
      \begin{itemize}
        \item Where occupied space is
        \item Where the route should go
        \item Whether the forward corridor is blocked
        \item Which side has more clearance
        \item Whether there is enough space to recover
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{What the policy does NOT use:}
      \begin{itemize}
        \item Object category labels (``sheep'', ``cone'', ``car'')
        \item Scenario cluster IDs
        \item Memorised trajectory patterns
        \item Any AV training data
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Formal Invariance Claim}
  Let scene $s$ define:
  \[
    G(s) = \text{occupancy field} \qquad R(s) = \text{route geometry}
  \]
  Let $T$ be any relabelling of object identities, scenario names, or cluster tags
  that leaves $G$ and $R$ unchanged.

  \[
    \boxed{M(s) = M(T(s)) \quad \forall\, s,\; \forall\, T \text{ s.t.\ } G(T(s))=G(s),\; R(T(s))=R(s)}
  \]

  \begin{itemize}
    \item \texttt{"sheep"} $\to$ \texttt{"debris"} changes nothing
    \item Cluster ID changes nothing
    \item Object label changes nothing
    \item \textbf{Only physical geometry enters the decision}
  \end{itemize}

  \vspace{0.5em}
  \begin{exampleblock}{This is falsifiable — and tested}
    A regression suite relabels every object in the scene and asserts the selected
    maneuver, combined score, and decision reason are identical.
    \textbf{It passes.}
  \end{exampleblock}
\end{frame}

\begin{frame}{Spotlight Reflex — It Works}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\linewidth]{images/kf_spotlight_success.png}
    \end{column}
    \begin{column}{0.45\textwidth}
      \vspace{0.4em}
      \textbf{Same scenario, same seed as the baseline failure.} \\[0.4em]
      \textbf{Policy:} Spotlight Reflex \\[0.4em]
      \textbf{Result:} selects \code{evasive\_right}, clears the actor, returns to lane centre.
      \vspace{0.4em}
      \textbf{How:}
      \begin{itemize}
        \item \code{obstacle\_pressure} rising on the left
        \item \code{right\_clearance} $>$ \code{left\_clearance}
        \item Escape side = right
        \item Geometry resolved the decision — no pattern match to a memorised scene
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Policy Architecture}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Six World-State Scalars}
  All signals computed from raw obstacle positions and route geometry.
  \textbf{Nothing is learned here; nothing is scene-specific.}

  \vspace{0.4em}
  \begin{center}
  \begin{tabular}{lll}
    \toprule
    \textbf{Signal} & \textbf{Formula / Definition} & \textbf{Range} \\
    \midrule
    \code{obstacle\_pressure} & $p = \text{clip}\!\left(\frac{10-d_{\min}}{10},\,0,\,1\right)$ & $[0,1]$ \\[4pt]
    \code{route\_blockage}    & fraction of forward corridor occupied & $[0,1]$ \\[4pt]
    \code{corridor\_blocked}  & hard blockage within stopping distance & bool \\[4pt]
    \code{left\_clearance}    & free lateral space to the left (m) & $\geq 0$ \\[4pt]
    \code{right\_clearance}   & free lateral space to the right (m) & $\geq 0$ \\[4pt]
    \code{preferred\_escape\_side} & $\argmax(\text{left},\,\text{right})$ & L / R \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \small $d_{\min}$ = distance to nearest obstacle in metres.
  Pressure saturates to 1.0 at contact and is zero beyond 10 m.
\end{frame}

\begin{frame}{Nine Maneuver Candidates}
  Each candidate is a \textbf{20-point, 5-second trajectory at 4 Hz}
  in vehicle coordinates, generated by Hermite smoothstep interpolation.

  \vspace{0.3em}
  \begin{center}
  \small
  \begin{tabular}{llrrl}
    \toprule
    \textbf{Maneuver} & \textbf{Intent} & \textbf{Speed scale} & \textbf{Lateral offset} & \textbf{Profile} \\
    \midrule
    \code{stop}          & Emergency stop     & 0.0  & 0.0 m   & — \\
    \code{crawl}         & Very slow progress & 0.15 & 0.0 m   & smooth \\
    \code{maintain}      & Hold speed         & 1.0  & 0.0 m   & smooth \\
    \code{slow\_yield}   & Decelerate, yield  & 0.5  & 0.0 m   & smooth \\
    \code{nudge\_left}   & Minor left shift   & 0.8  & +0.9 m  & smooth \\
    \code{nudge\_right}  & Minor right shift  & 0.8  & −0.9 m  & smooth \\
    \code{evasive\_left} & Strong left evasion  & 0.7 & +2.2 m & \textbf{early} \\
    \code{evasive\_right}& Strong right evasion & 0.7 & −2.2 m & \textbf{early} \\
    \code{lane\_recover} & Return to centre   & 0.9  & 0.0 m   & smooth \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.2em}
  \small\textit{Early profile}: offset applied in first third of trajectory — ensures fast
  lateral separation in evasive scenarios.
\end{frame}

\begin{frame}{Trust-Region Scoring}
  Each candidate is scored against \textbf{pseudo-rater references} at two time horizons.
  A trust region defines where the candidate must land to earn the reference's score.

  \vspace{0.3em}
  \textbf{Speed-scaled tolerance:}
  \[
    s(v) = \text{clip}\!\left(\frac{v - 1.4}{11.0 - 1.4},\; 0,\; 1\right)\times 0.5 + 0.5
    \qquad [v \text{ in m/s}]
  \]
  Tighter at low speed ($s=0.5$ at $v=1.4$~m/s), wider at high speed ($s=1.0$ at $v=11$~m/s).

  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \vspace{0.3em}
      \begin{tabular}{lcc}
        \toprule
        \textbf{Horizon} & \textbf{Lateral} & \textbf{Longitudinal} \\
        \midrule
        3 s & $1.0\times s(v)$ m & $4.0\times s(v)$ m \\
        5 s & $1.8\times s(v)$ m & $7.2\times s(v)$ m \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.55\textwidth}
      \[
        \text{score}_t(\tau,r) =
        \begin{cases}
          r.\text{score} & \delta = 0 \\
          \max\!\bigl(r.\text{score}\cdot 0.1^\delta,\; 4\bigr) & \delta > 0
        \end{cases}
      \]
      \[
        S(\tau) = 0.5\cdot\max_r\text{score}_{3s} + 0.5\cdot\max_r\text{score}_{5s}
      \]
      \small $\delta$ = normalised overshoot beyond the region.\\
      Floor of 4 corresponds to an unsafe rating.
    \end{column}
  \end{columns}
\end{frame}

  \begin{center}
  \begin{tikzpicture}[

\section{Simulation: What We Built and How It Performed}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{The Simulator}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Built from scratch — no dependency on AlpaSim or Waymo internals.}

      \vspace{0.4em}
      \begin{itemize}
        \item \textbf{Closed-loop 2D engine:} ego + actors step simultaneously;
              policy must react in real time — no lookahead
        \item \textbf{11 WOD-E2E clusters:} construction, spotlight, FOD,
              pedestrian, cyclist, cut-in, animal, intersection, special vehicle, stopped, others
        \item \textbf{Compositional OOD:} 8 topologies $\times$ 11 hazard modules
              $\times$ 4 weather modes — axes sampled \emph{independently}
        \item \textbf{8 actor behaviour models:} linear, cut-in, swerve, darting,
              erratic pedestrian, sudden brake, hesitating, wrong-way
        \item \textbf{COMPASS benchmark:} composite score across oracle quality,
              geometric reasoning, recovery, generalisation gap; threshold 7.0/10
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \includegraphics[width=\linewidth]{images/kf_construction.png}
      \vspace{0.2em}
      \small\textit{Construction zone — 5.2\,m corridor, cone field, lane closure.\\
      Policy selects \code{nudge\_right}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Compositional OOD: The Independence Property}
  Standard simulators replay fixed scenario clusters.
  If you test on the same clusters you designed for, \textbf{you cannot claim generalisation}.

  \vspace{0.3em}
  Our generator makes topology and hazard \textbf{statistically independent}:
  \[
    P(\text{topology},\; \text{hazard}) = P(\text{topology})\;\times\; P(\text{hazard})
  \]

  \vspace{0.2em}
  \begin{center}
  \begin{tikzpicture}[
    node distance=0.3cm and 0.5cm,
    ax/.style={rectangle, rounded corners=3pt, draw=spotTeal!50, fill=spotTeal!10,
               text=textLight, font=\scriptsize, align=center,
               minimum width=2.4cm, minimum height=0.9cm},
    sc/.style={rectangle, rounded corners=4pt, draw=spotGold, fill=spotGold!15,
               text=textLight, font=\small\bfseries, align=center,
               minimum width=2.8cm, minimum height=1.0cm},
    arr/.style={-Stealth, thick, color=spotGray}
  ]
    \node[ax] (T) {8 Topologies\\straight · curve\\roundabout · …};
    \node[ax, right=3.2cm of T] (S) [sc] {Scenario\\full geometry\\+ actors};
    \node[ax, above=0.2cm of T, xshift=-0.2cm] (H)  {11 Hazards\\construction · FOD\\spotlight · …};
    \node[ax, below=0.2cm of T, xshift=-0.2cm] (W)  {4 Weather\\clear · rain\\fog · night};
    \node[ax, below=0.8cm of W, xshift=0.2cm]  (O)  {Novel Objects\\cone · animal\\debris · barrier};

  \vspace{0.2em}
  A construction hazard can appear in a \textbf{roundabout in fog}.
  A pedestrian can appear on an \textbf{S-curve at night}.
  No cluster label tells the policy what geometry or weather it will face next.
\end{frame}

\begin{frame}{Gauntlet: Head-to-Head Comparison}
  Same \textbf{420 scenarios}, matched seeds 1–80, four topology types.
  Same geometry, same actors, same timing for both policies.

  \vspace{0.5em}
  \begin{center}
  \begin{tabular}{lcc}
    \toprule
    \textbf{Policy} & \textbf{Pass rate} & \textbf{Collision rate} \\
    \midrule
    Baseline (trajectory extrapolation, no world-state) & 2.1\% (9/420) & \bad{20.5\% (86)} \\
    \textbf{Spotlight Reflex} & \good{\textbf{57.6\% (242/420)}} & \good{\textbf{7.9\% (33)}} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{27$\times$ more passes.}\\
      \textbf{2.6$\times$ fewer collisions.}\\[0.4em]
      The baseline collides in 1 in 5 gauntlet runs and passes nearly none.
    \end{column}
    \begin{column}{0.48\textwidth}

\section{WOD-E2E Benchmark}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{What Is WOD-E2E and How It Is Scored}
  \textbf{Waymo Open Dataset End-to-End Driving:}
  \begin{itemize}
    \item 479 validation frames from real Waymo sensor recordings
    \item 11 long-tail scenario types — same clusters as our simulator
    \item For each frame: Waymo human raters expressed pairwise preferences between
          candidate trajectories
    \item \textbf{RFS (Rater Feedback Score):} derived from those pairwise preferences —
          how often the selected trajectory was preferred. Higher is better.
  \end{itemize}

  \vspace{0.4em}
  \textbf{Evaluation protocol — 5-fold segment-grouped cross-validation:}
  \[
    \text{RFS}_{5\text{-fold}} = \frac{1}{5}\sum_{k=1}^{5}\text{RFS}(\text{fold}\;k\;\text{held-out})
  \]
  Frames from the same driving segment stay in the same fold — prevents leakage.
  Each fold: $\sim$383 train / $\sim$96 test frames. Fold std $\approx 0.19$, 95\% CI $\approx\pm0.17$.

  \vspace{0.3em}
  \begin{alertblock}{Two backends — not interchangeable}
    Local (ours): CV baseline 7.131 \quad Official (Waymo): CV baseline 7.022 \\
    All development numbers use the local backend.
    2-fold and 5-fold results are \emph{not} directly comparable.
  \end{alertblock}
\end{frame}

\begin{frame}{Appendix: Random Fourier Features — Derivation}
  \textbf{Why RFF?} A standard ridge classifier is linear. The preference signal requires a
  non-linear separator, which normally needs an $N\times N$ kernel matrix — too slow for CV.

  RFF approximates the Gaussian kernel \emph{without the matrix} via a random projection:
  \begin{align*}
    \mathbf{W} &\sim \mathcal{N}(0,\,\sigma^{-2}\mathbf{I}_{D\times p})\;\text{(fixed at fit time)} \qquad D=512,\;\sigma=7.858 \\
    \mathbf{b} &\sim \text{Uniform}([0, 2\pi]^D) \\
    \boldsymbol{\varphi}(\mathbf{x}) &= \sqrt{\tfrac{2}{D}}\cdot\cos(\mathbf{W}^\top\tilde{\mathbf{x}}+\mathbf{b}) \\
    \mathbb{E}\bigl[\boldsymbol{\varphi}(\mathbf{x})^\top\boldsymbol{\varphi}(\mathbf{x}')\bigr]
    &= \exp\!\left(-\tfrac{\|\mathbf{x}-\mathbf{x}'\|^2}{2\sigma^2}\right)
    \quad\text{(Bochner's theorem)}
  \end{align*}
  Only the ridge weights $\mathbf{w}$ (512-dim) are learned: $\hat{y} = \mathbf{w}^\top\boldsymbol{\varphi}(\mathbf{x})+b$.
  Cost: $O(Dp)$ per inference instead of $O(N)$.
\end{frame}

\begin{frame}{Appendix: Optuna TPE Algorithm}
  \textbf{Optuna} uses Tree-structured Parzen Estimators (TPE) — Bayesian optimisation that
  is more sample-efficient than random search.

  After $N$ trials with observed objective values:
  \begin{align*}
    \text{Good trials}\; G &= \{x : f(x) < f^*\} \quad\text{(top $p$-percentile)} \\
    \text{Bad trials}\;  B &= \{x : f(x) \geq f^*\} \\
    \ell(x) &= P(x \mid x \in G)\qquad \text{fit with kernel density estimation} \\
    g(x) &= P(x \mid x \in B) \\
    x_\text{next} &= \argmax_x\;\frac{\ell(x)}{g(x)}
    \quad\text{(exploit good configs, avoid bad ones)}
  \end{align*}

  \vspace{0.3em}
  \textbf{Our GPU MLP search (20 trials):}
  \begin{itemize}
    \item Best: trial\_0004 → $h=64$, lr=0.003866, batch=512, 10 epochs → \good{7.845 RFS}
    \item Mean of all 20 trials: 7.769; only 1 beat the RFF baseline (7.834)
    \item Lesson: with only 479 frames and 5 folds, Optuna can exploit fold-specific patterns
  \end{itemize}
\end{frame}
